4. TypeScript 类型定义

所有类型集中定义在 src/types/interview.ts。

// ===== 岗位 =====
interface Position {
  id: string
  title: string           // "Java 后端开发工程师"
  experience: string      // "3-5 年经验"
  city: string            // "北京"
  industry: string        // "互联网"
}

// ===== 简历 =====
interface Resume {
  id: string
  fileName: string
  fileType: 'pdf' | 'docx' | 'image'
  updatedAt: string       // ISO date
  tags: string[]           // ["Java", "Spring Boot", "MySQL"]
  previewUrl?: string
}

// ===== 面试配置 =====
interface InterviewConfig {
  positionId: string
  resumeId?: string
  interviewerStyle: 'gentle' | 'standard' | 'pressure'
  difficulty: 'junior' | 'mid' | 'senior' | 'expert'
  scenario: 'technical' | 'project' | 'hr' | 'comprehensive'
  questionCount: 3 | 5 | 8
  smartHint: boolean
  muteMode: boolean
}

// ===== 对话消息（4 种类型） =====
type MessageRole = 'question' | 'user' | 'analysis' | 'ai'

interface ChatMessage {
  id: string
  role: MessageRole
  content: string
  timestamp: number
  // 问题类型专有
  questionIndex?: number
  questionTotal?: number
  source?: string          // "题库随机" / "AI 追问"
  // AI 解答专有：可能含代码块
  codeBlocks?: CodeBlock[]
  // 实时分析专有
  highlights?: string[]
  gaps?: string[]
  dimensionScores?: DimensionScore[]
}

interface CodeBlock {
  language: string         // "Python" / "Java" / "JavaScript"
  code: string
}

// ===== 维度评分 =====
interface DimensionScore {
  key: 'relevance' | 'professionalism' | 'fluency' | 'interactivity' | 'confidence' | 'logic'
  label: string            // "回答相关性" / "专业度" / "表达流畅度" ...
  score: number            // 0-100
}

// ===== 题目进度 =====
interface QuestionProgress {
  index: number
  title: string            // "自我介绍" / "算法题：爬楼梯"
  status: 'completed' | 'active' | 'pending'
}

// ===== 复盘报告 =====
interface InterviewReport {
  id: string
  summary: string          // 面试概要文字
  radarScores: DimensionScore[]
  pros: ReportItem[]        // 优点列表
  cons: ReportItem[]        // 缺点列表
  analysisList: AnalysisItem[]
  interviewerComment: string
  suggestions: Suggestion[]
  knowledgeCards: KnowledgeCard[]
}

interface ReportItem {
  id: string
  title: string
  quote: string            // 面试对话中的锚点引用原文
  relatedMessageId?: string // 可点击跳回对应对话位置
}

interface AnalysisItem {
  id: string
  title: string
  content: string
  score: number            // 0-100
  type: 'strong' | 'weak'
}

interface Suggestion {
  index: number
  title: string
  content: string
}

interface KnowledgeCard {
  id: string
  title: string
  description: string
  linkText: string         // "查看相关题目 →"
  linkUrl: string
}

6. 路由设计

// router/index.ts
const routes = [
  {
    path: '/',
    name: 'InterviewHome',
    component: () => import('@/views/InterviewHome.vue'),
    meta: { title: '面试学习中心' },
  },
  {
    path: '/interview/prep',
    name: 'InterviewPrep',
    component: () => import('@/views/InterviewPrep.vue'),
    meta: { title: '面试准备' },
  },
  {
    path: '/interview/session',
    name: 'InterviewSession',
    component: () => import('@/views/InterviewSession.vue'),
    meta: { title: '面试进行中' },
  },
  {
    path: '/interview/report/:id',
    name: 'InterviewReport',
    component: () => import('@/views/InterviewReport.vue'),
    meta: { title: '面试复盘报告' },
    props: true,
  },
]

// 全局路由守卫：未配置岗位时不能进入面试进行页
router.beforeEach((to, from, next) => {
  const store = useInterviewStore()
  if (to.name === 'InterviewSession' && !store.selectedPosition) {
    next({ name: 'InterviewPrep' })
  } else {
    next()
  }
})

12. API 接口契约（Mock 阶段）

MVP 阶段使用 mock/ 目录下的静态数据模拟，接口签名如下，后续对接真实后端时只需替换实现：

// api/interview.ts

/** 获取岗位列表 */
export async function getPositions(): Promise<Position[]>

/** 上传简历，返回解析结果 */
export async function uploadResume(file: File): Promise<Resume>

/** 获取用户在线简历 */
export async function getOnlineResume(): Promise<Resume | null>

/** 创建面试会话 */
export async function createSession(config: InterviewConfig): Promise<{ sessionId: string }>

/** 提交用户回答，获取 AI 面试官的下一条消息 */
export async function submitAnswer(
  sessionId: string,
  answer: string
): Promise<{
  aiReply: ChatMessage        // 面试官回复/解答
  analysis: ChatMessage       // 实时分析
  dimensions: DimensionScore[] // 更新后的维度评分
  nextQuestion?: ChatMessage   // 如果是下一题则返回
}>

/** 结束面试，触发报告生成 */
export async function endSession(sessionId: string): Promise<{ reportId: string }>

/** 获取复盘报告 */
export async function getReport(reportId: string): Promise<InterviewReport>

给 AI 编码工具的提示：
